public CalcResult CalculateSegment( PipeSegment seg, FluidProperties fluid) { double area = seg.AreaM2; double v = (area > 0.0) ? seg.FlowM3s / area : 0.0; double re = fluid.Density * Math.Abs(v) * seg.DiameterM / fluid.DynamicViscosity; double rough = MaterialDatabase .GetRoughness(seg.MaterialKey); // Darcy–Weisbach + Colebrook–White iteration double f = (re > 0.0) ? FrictionFactor.Calculate(re, rough, seg.DiameterM) : 0.0; double dpPerL = f * (1.0 / seg.DiameterM) * fluid.Density * v * v / 2.0; double singular = seg.Fittings .Sum(fit => fit.CalculateLossPa( Math.Abs(v), fluid.Density)); seg.VelocityMs = v; seg.FrictionPaPm = dpPerL; seg.TotalLossPa = dpPerL * seg.LengthM + singular; }

function filterHtml(html, keywords) { const doc = new DOMParser() .parseFromString(html, 'text/html'); function processNode(node) { if (node.nodeType === Node.TEXT_NODE) { const text = node.textContent.toLowerCase(); // Requires ALL keywords to be present return keywords.every( kw => text.includes(kw.toLowerCase()) ); } if (node.nodeType === Node.ELEMENT_NODE) { let hasRelevant = false; Array.from(node.childNodes).reverse() .forEach(child => { if (!processNode(child)) node.removeChild(child); else hasRelevant = true; }); if (hasRelevant) highlightKeywords(node, keywords); return hasRelevant; } return true; } processNode(doc.body); return doc.body.innerHTML; }
